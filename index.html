    <label>Veilige Z-hoogte (mm): <input type="number" id="safeZ" value="5" step="0.5"></label>

    <label>Freesdiepte Z (mm): <input type="number" id="cutZ" value="-1" step="0.1"></label>

    <label>Breedte Werkstuk (mm): <input type="number" id="targetWidth" value="100"></label>

    <label>Freessnelheid F (mm/min): <input type="number" id="feedRate" value="500"></label>

  <button onclick="generateGCode()">💾 Genereer & Download G-Code</button>

 

  <script>

    const imageInput = document.getElementById('imageInput');

    const canvas = document.getElementById('previewCanvas');

    const ctx = canvas.getContext('2d');

    let img = new Image();

 

    imageInput.addEventListener('change', (e) => {

      const file = e.target.files[0];

      if (!file) return;

      const reader = new FileReader();

      reader.onload = (event) => {

        img.onload = () => {

          canvas.width = img.width;

          canvas.height = img.height;

          ctx.drawImage(img, 0, 0);

        };

        img.src = event.target.result;

      };

      reader.readAsDataURL(file);

    });

 

    function generateGCode() {

      if (!img.src) { alert("Upload eerst een foto!"); return; }

 

      const safeZ = parseFloat(document.getElementById('safeZ').value);

      const cutZ = parseFloat(document.getElementById('cutZ').value);

      const targetWidth = parseFloat(document.getElementById('targetWidth').value);

      const feedRate = parseFloat(document.getElementById('feedRate').value);

      const thresh = parseInt(document.getElementById('threshold').value);

 

      const scale = targetWidth / canvas.width;

      const imgData = ctx.getImageData(0, 0, canvas.width, canvas.height);

      const data = imgData.data;

 

      let gcode = [];

      gcode.push("; GRBL Contour Output (Android)");

      gcode.push("G21 ; mm");

      gcode.push("G90 ; Absoluut");

      gcode.push(`G0 Z${safeZ}`);

      gcode.push("M3 S10000");

 

      for (let y = 0; y < canvas.height; y += 2) {

        let cutting = false;

        for (let x = 0; x < canvas.width; x++) {

          const idx = (y * canvas.width + x) * 4;

          const avg = (data[idx] + data[idx + 1] + data[idx + 2]) / 3;

 

          if (avg < thresh) {

            const realX = (x * scale).toFixed(3);

            const realY = ((canvas.height - y) * scale).toFixed(3);

 

            if (!cutting) {

              gcode.push(`G0 X${realX} Y${realY}`);

              gcode.push(`G1 Z${cutZ} F150`);

              cutting = true;

            } else {

              gcode.push(`G1 X${realX} Y${realY} F${feedRate}`);

            }

          } else {

            if (cutting) {

              gcode.push(`G0 Z${safeZ}`);

              cutting = false;

            }

          }

        }

        if (cutting) {

          gcode.push(`G0 Z${safeZ}`);

          cutting = false;

        }

      }

 

      gcode.push(`G0 Z${safeZ}`);

      gcode.push("G0 X0 Y0");

      gcode.push("M5");

      gcode.push("M30");

 

      const blob = new Blob([gcode.join('\n')], { type: 'text/plain' });

      const a = document.createElement('a');

      a.href = URL.createObjectURL(blob);

      a.download = 'android_cnc_output.gcode';

      a.click();

    }

  </script>
