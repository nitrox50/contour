  <script>

    const imageInput = document.getElementById('imageInput');

    const canvas = document.getElementById('previewCanvas');

    const ctx = canvas.getContext('2d');

    const hmCanvas = document.getElementById('heightmapCanvas');

    const hmCtx = hmCanvas.getContext('2d');

    let img = new Image();

 

    const defaultLibrary = [

      { id: '1', cat: 'vlak', name: 'Vlakfrees 3.175mm (1/8")', dia: 3.175, angle: 0, feed: 600, spindle: 12000, stepover: 40 },

      { id: '2', cat: 'vlak', name: 'Vlakfrees 6.0mm', dia: 6.0, angle: 0, feed: 1000, spindle: 10000, stepover: 50 },

      { id: '3', cat: 'bol',  name: 'Ballnose Bolfrees 2.0mm', dia: 2.0, angle: 0, feed: 400, spindle: 14000, stepover: 15 },

      { id: '4', cat: 'punt', name: 'V-Bit Puntfrees 60° (6mm)', dia: 6.0, angle: 60, feed: 500, spindle: 12000, stepover: 25 },

      { id: '5', cat: 'punt', name: 'V-Bit Puntfrees 90° (6mm)', dia: 6.0, angle: 90, feed: 500, spindle: 12000, stepover: 30 }

    ];

 

    let bitLibrary = JSON.parse(localStorage.getItem('cnc_bit_library')) || defaultLibrary;

 

    function renderBitSelect() {

      const select = document.getElementById('bitSelect');

      select.innerHTML = '';

      const categories = { vlak: '--- VLAKFREEZEN ---', bol: '--- BOLFREEZEN (BALLNOSE) ---', punt: '--- PUNTFREEZEN (V-BIT) ---' };

 

      for (let cat in categories) {

        let group = document.createElement('optgroup');

        group.label = categories[cat];

        bitLibrary.filter(b => b.cat === cat).forEach(bit => {

          let opt = document.createElement('option');

          opt.value = bit.id;

          opt.textContent = bit.name;

          group.appendChild(opt);

        });

        select.appendChild(group);

      }

      loadSelectedBit();

    }

 

    function loadSelectedBit() {

      const id = document.getElementById('bitSelect').value;

      const bit = bitLibrary.find(b => b.id === id);

      if (bit) {

        document.getElementById('feedRate').value = bit.feed;

        document.getElementById('spindleSpeed').value = bit.spindle;

        document.getElementById('stepover').value = bit.stepover;

        document.getElementById('bitAngle').value = bit.angle || 0;

      }

    }

 

    function toggleAddBitModal() {

      const modal = document.getElementById('addBitModal');

      modal.style.display = modal.style.display === 'block' ? 'none' : 'block';

    }

 

    function saveNewBit() {

      const cat = document.getElementById('newCategory').value;

      const name = document.getElementById('newName').value || 'Custom Frees';

      const dia = parseFloat(document.getElementById('newDia').value);

      const angle = parseFloat(document.getElementById('newAngle').value) || 0;

      const feed = parseInt(document.getElementById('newFeed').value);

      const spindle = parseInt(document.getElementById('newSpindle').value);

      const stepover = parseInt(document.getElementById('newStepover').value);

 

      const newBit = { id: Date.now().toString(), cat, name, dia, angle, feed, spindle, stepover };

      bitLibrary.push(newBit);

      localStorage.setItem('cnc_bit_library', JSON.stringify(bitLibrary));

     

      renderBitSelect();

      document.getElementById('bitSelect').value = newBit.id;

      loadSelectedBit();

      toggleAddBitModal();

    }

 

    imageInput.addEventListener('change', (e) => {

      const file = e.target.files[0];

      if (!file) return;

      const reader = new FileReader();

      reader.onload = (event) => {

        img.onload = () => {

          canvas.width = img.width;

          canvas.height = img.height;

          hmCanvas.width = img.width;

          hmCanvas.height = img.height;

          updatePreviews();

        };

        img.src = event.target.result;

      };

      reader.readAsDataURL(file);

    });

 

    document.getElementById('threshold').addEventListener('input', () => { if (img.src) updatePreviews(); });

 

    // Hulpfunctie voor een thermische regenboog-kleurenschaal (Jet Colormap)

    function getJetColor(val) {

      // val is 0.0 (diepste) tot 1.0 (hoogste)

      let r = Math.max(0, Math.min(255, Math.round(255 * (1.5 - Math.abs(val * 4 - 3)))));

      let g = Math.max(0, Math.min(255, Math.round(255 * (1.5 - Math.abs(val * 4 - 2)))));

      let b = Math.max(0, Math.min(255, Math.round(255 * (1.5 - Math.abs(val * 4 - 1)))));

      return [r, g, b];

    }

 

    function updatePreviews() {

      ctx.drawImage(img, 0, 0);

      const imgData = ctx.getImageData(0, 0, canvas.width, canvas.height);

      const data = imgData.data;

      const thresh = parseInt(document.getElementById('threshold').value);

 

      const hmData = hmCtx.createImageData(canvas.width, canvas.height);

      const hmPixels = hmData.data;

 

      for (let i = 0; i < data.length; i += 4) {

        const avg = (data[i] + data[i + 1] + data[i + 2]) / 3;

 

        // Contour overlay (Rood op eerste canvas)

        if (avg < thresh) {

          data[i] = 255; data[i+1] = 0; data[i+2] = 0;

        }

 

        // False Color Heatmap (Regenboog)

        const norm = avg / 255;

        const [r, g, b] = getJetColor(norm);

 

        hmPixels[i] = r;

        hmPixels[i+1] = g;

        hmPixels[i+2] = b;

        hmPixels[i+3] = 255;

      }

 

      ctx.putImageData(imgData, 0, 0);

      hmCtx.putImageData(hmData, 0, 0);

    }

 

    function generateGCode() {

      if (!img.src) { alert("Upload eerst een afbeelding!"); return; }

 

      const bitId = document.getElementById('bitSelect').value;

      const bit = bitLibrary.find(b => b.id === bitId);

      const stepoverPct = parseFloat(document.getElementById('stepover').value);

      const targetWidth = parseFloat(document.getElementById('targetWidth').value);

      const safeZ = parseFloat(document.getElementById('safeZ').value);

      const cutZ = parseFloat(document.getElementById('cutZ').value);

      const feedRate = parseFloat(document.getElementById('feedRate').value);

      const spindle = parseInt(document.getElementById('spindleSpeed').value);

      const thresh = parseInt(document.getElementById('threshold').value);

 

      let effectiveDia = bit.dia;

      if (bit.cat === 'punt' && bit.angle > 0) {

        const rad = (bit.angle / 2) * (Math.PI / 180);

        effectiveDia = Math.min(bit.dia, 2 * Math.abs(cutZ) * Math.tan(rad));

      }

 

      const stepmm = Math.max(0.05, effectiveDia * (1 - (stepoverPct / 100)));

      const scale = targetWidth / canvas.width;

      const stepPx = Math.max(1, Math.round(stepmm / scale));

 

      ctx.drawImage(img, 0, 0);

      const imgData = ctx.getImageData(0, 0, canvas.width, canvas.height);

      const data = imgData.data;

 

      let gcode = [];

      gcode.push(`; GRBL Output - ${bit.name} (${bit.cat.toUpperCase()})`);

      gcode.push(`; Eff. Dia: ${effectiveDia.toFixed(3)}mm, Step: ${stepmm.toFixed(3)}mm (${stepPx}px)`);

      gcode.push("G21 ; mm");

      gcode.push("G90 ; Absoluut");

      gcode.push(`G0 Z${safeZ}`);

      gcode.push(`M3 S${spindle}`);

 

      for (let y = 0; y < canvas.height; y += stepPx) {

        let cutting = false;

        for (let x = 0; x < canvas.width; x += stepPx) {

          const idx = (y * canvas.width + x) * 4;

          const avg = (data[idx] + data[idx + 1] + data[idx + 2]) / 3;

 

          if (avg < thresh) {

            const realX = (x * scale).toFixed(3);

            const realY = ((canvas.height - y) * scale).toFixed(3);

 

            if (!cutting) {

              gcode.push(`G0 X${realX} Y${realY}`);

              gcode.push(`G1 Z${cutZ} F150`);

              cutting = true;

            } else {

              gcode.push(`G1 X${realX} Y${realY} F${feedRate}`);

            }

          } else {

            if (cutting) {

              gcode.push(`G0 Z${safeZ}`);

              cutting = false;

            }

          }

        }

        if (cutting) {

          gcode.push(`G0 Z${safeZ}`);

          cutting = false;

        }

      }

 

      gcode.push(`G0 Z${safeZ}`);

      gcode.push("G0 X0 Y0\nM5\nM30");

 

      const resultText = gcode.join('\n');

      document.getElementById('gcodePreview').value = resultText;

      updatePreviews();

 

      const blob = new Blob([resultText], { type: 'text/plain' });

      const a = document.createElement('a');

      a.href = URL.createObjectURL(blob);

      a.download = `cnc_${bit.cat}_${effectiveDia.toFixed(1)}mm.gcode`;

      a.click();

    }

 

    renderBitSelect();

  </script>
